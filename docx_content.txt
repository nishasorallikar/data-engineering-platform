Find employees who have the highest salary in each department.

SQL Solution
SELECT
    d.name AS Department,
    e.name AS Employee,
    e.salary AS Salary
FROM Employee e
JOIN Department d
    ON e.departmentId = d.id
JOIN (
    SELECT departmentId, MAX(salary) AS max_salary
    FROM Employee
    GROUP BY departmentId
) m

    max_salary,
    (employee_df.departmentId == max_salary.departmentId) &
    (employee_df.salary == max_salary.max_salary)
).join(
    department_df,
    employee_df.departmentId == department_df.id
).select(
    department_df.name.alias("Department"),
    employee_df.name.alias("Employee"),
    employee_df.salary.alias("Salary")
)

WITH ranked AS (

Day 23 - Game Play Analysis IV (#550)
Question
Find fraction of players who logged in again the day after first login.
Explanation
Uses next/previous row logic (LEAD/LAG or joins).

Very common in retention analysis.
Input Table
player_id
event_date
1
2024-01-01
1
2024-01-02
2
2024-01-03
3
2024-01-05
3
2024-01-06
fraction
0.67
SQL Solution
WITH first_login AS (
    SELECT
        player_id,
        MIN(event_date) AS first_date
    FROM Activity
    GROUP BY player_id
)
SELECT ROUND(
    COUNT(DISTINCT a.player_id) /
    (SELECT COUNT(DISTINCT player_id) FROM Activity),
    2
) AS fraction
FROM Activity a
JOIN first_login f
ON a.player_id = f.player_id
WHERE DATEDIFF(a.event_date, f.first_date) = 1;
PySpark Solution
from pyspark.sql.functions import min, countDistinct, datediff, round, col

first_login = activity_df.groupBy("player_id").agg(min("event_date").alias("first_date"))

retained = activity_df.join(first_login, "player_id").filter(datediff(col("event_date"), col("first_date")) == 1)

result = retained.agg(round(countDistinct("player_id") / activity_df.select("player_id").distinct().count(),2).alias("fraction"))

Day 76 - Game Play Analysis V (#1532)
Problem Statement
Calculate player retention based on consecutive login dates.
Explanation
Uses:
- Date arithmetic
- LEAD/LAG
- Retention logic
Input Table: Activity
player_id
event_date
1
2024-01-01
1
2024-01-02
2
2024-01-01
Expected Output
fraction
0.50
SQL Solution
WITH first_login AS (
    SELECT
        player_id,
        MIN(event_date) AS first_date
    FROM Activity
    GROUP BY player_id
)
SELECT
    ROUND(COUNT(DISTINCT a.player_id) / (SELECT COUNT(DISTINCT player_id) FROM Activity),2) AS fraction
FROM Activity a
JOIN first_login f
ON a.player_id = f.player_id
AND a.event_date = DATE_ADD(f.first_date, INTERVAL 1 DAY);
PySpark Solution
from pyspark.sql.functions import min, date_add, countDistinct, col

first_login = activity_df.groupBy("player_id").agg(min("event_date").alias("first_date"))

retained = activity_df.join(first_login,"player_id").filter(col("event_date") == date_add(col("first_date"), 1))

result = retained.agg((countDistinct("player_id") / activity_df.select("player_id").distinct().count()).alias("fraction"))
